async function loadEmployeeDocuments(employeeId){
  const context=employeeDocumentContext(),ticket=++employeeDocumentUi.request;
  const result=await api(`/api/employees/${Number(employeeId)}/documents`,{timeoutMs:15000,silentStatus:true});
  if(!employeeDocumentContextValid(context)||ticket!==employeeDocumentUi.request||Number(employeeId)!==employeeDocumentUi.employeeId)return result;
  if(!Array.isArray(result.data))throw new Error('ข้อมูลเอกสารไม่ครบ กรุณาลองโหลดใหม่');
  employeeDocumentUi.docs=result.data;employeeDocumentUi.permissions=result.permissions||{};
  employeeDocumentRenderList();employeeDocumentUpdateCount();
  if(!result.management_version)employeeDocumentSetMessage('Backend ยังไม่เป็น P10.4 · ยังดูไฟล์ได้ แต่ต้องอัป Backend ก่อนใช้แก้ไข/กู้คืน',true);
  return result;
}

async function uploadEmployeeDocument(){
  if(employeeDocumentUi.busy||!employeeDocumentUi.permissions.can_edit)return;
  const file=$('#employeeDocumentFile').files?.[0];if(!file)return employeeDocumentSetMessage('กรุณาเลือกไฟล์',true);
  if(file.size>10*1024*1024)return employeeDocumentSetMessage('ไฟล์ต้องไม่เกิน 10 MB',true);
  if(!/\.(pdf|png|jpe?g|webp|docx?)$/i.test(file.name))return employeeDocumentSetMessage('รองรับ PDF รูปภาพ PNG/JPG/WEBP และ Word เท่านั้น',true);
  if($('#employeeDocumentExpires').value&&$('#employeeDocumentDate').value&&$('#employeeDocumentExpires').value<$('#employeeDocumentDate').value)return employeeDocumentSetMessage('วันหมดอายุต้องไม่ก่อนวันที่เอกสาร',true);
  const context=employeeDocumentContext(),employeeId=employeeDocumentUi.employeeId,status=$('#employeeDocumentUploadStatus');employeeDocumentBusy(true);status.textContent=`กำลังอัปโหลด ${file.name}…`;employeeDocumentSetMessage('');
  let saved=false;
  try{
    const form=new FormData();form.append('file',file);for(const [key,id] of Object.entries({title:'employeeDocumentTitle',document_type:'employeeDocumentType',document_date:'employeeDocumentDate',expires_at:'employeeDocumentExpires',visibility:'employeeDocumentVisibility',confidentiality:'employeeDocumentConfidentiality',note:'employeeDocumentNote'}))form.append(key,$('#'+id).value);
    await api(`/api/employees/${employeeId}/documents`,{method:'POST',body:form,timeoutMs:60000,silentStatus:true});saved=true;
    if(!employeeDocumentContextValid(context))return;
    $('#employeeDocumentFile').value='';$('#edUploadPanel').hidden=true;status.textContent='อัปโหลดสำเร็จ';employeeDocumentSetMessage('อัปโหลดสำเร็จ');
    try{await loadEmployeeDocuments(employeeId);}catch{if(employeeDocumentContextValid(context))employeeDocumentSetMessage('ไฟล์บันทึกแล้ว แต่โหลดรายการไม่สำเร็จ กด “โหลดรายการ” ไม่ต้องอัปไฟล์ซ้ำ',true);}
  }catch(error){if(employeeDocumentContextValid(context)){status.textContent=saved?'บันทึกแล้ว':'อัปโหลดไม่สำเร็จ';employeeDocumentSetMessage(employeeDocumentError(error),true);}}
  finally{employeeDocumentBusy(false);}
}
